clc;
clear all;
c=[6 4 1 5;8 9 2 7;4 3 6 4]
s=[14 16 5]
d=[6 10 15 4]
m=size(c,1) 
n=size(c,2)
if sum(s)==sum(d)
    fprintf("The given transportation problem is balanced")
    X=zeros(m,n)
    z=0
    Initialc=c
    for i=1:m
        for j=1:n
            cpq=min(c(:))
            if cpq==Inf
                break
            end
            [p1 q1]=find(cpq==c);
            xpq=min(s(p1),d(q1))
            [val ind]=max(xpq)
            p=p1(ind)
            q=q1(ind)
            X(p,q)=min(s(p),d(q))
            if min(s(p),d(q))==s(p)
                d(q)=d(q)-s(p)
                s(p)=s(p)-X(p,q)
                c(p,:)=Inf
            else
                s(p)=s(p)-d(q)
                d(q)=d(q)-X(p,q)
                c(:,q)=Inf
            end
        end
    end
    for i=1:m
        for j=1:n
            z=z+Initialc(i,j)*X(i,j)
        end
    end
    fprintf('The TP cost is %f',z)
    X
else
    fprintf("The given transportation problem is unbalanced");
end
